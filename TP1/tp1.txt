manual/current/functions/aggregating/#functions-collect

MATCH (p)-[:REVIEWED]->(m:Movie)
WHERE p.name = 'Jessica Thompson'
OPTIONAL MATCH (a)-[:ACTED_IN]->(m)
RETURN m.title AS title, collect(a.name) AS actors
ORDER BY title;

MATCH (p)-[:ACTED_IN]->(m:Movie)
RETURN p.name, COUNT(m) as number_movies ORDER BY number_movies DESC;

Question 2 — Prérequis
Quels sont les étudiants inscrits à un cours alors qu'ils ne sont inscrits à aucun de ses cours prérequis ?
MATCH (s:Student)-[:ENROLLED_IN]->(c:Course)
MATCH (prereq:Course)-[:PREREQUISITE_OF]->(c)
WHERE NOT EXISTS {
    MATCH (s)-[:ENROLLED_IN]->(prereq)
}
RETURN s.name AS student,
       c.title AS course,
       prereq.title AS missingPrerequisite;

Question 3 — Professeurs et départements

MATCH (p:Professor)-[:WORKS_IN]->(d:Department)

RETURN p.name AS professor,
       d.name AS worksIn,
       c.title AS course,

Question 5 — Analyse des départements
Quel département possède le plus d'étudiants inscrits à ses cours ? Afficher le département, le nombre d'étudiants distincts et les cours concernés.
MATCH (d:Department)<-[:OFFERED_BY]-(c:Course)
      <-[:ENROLLED_IN]-(s:Student)

RETURN d.name AS department,
       COUNT(DISTINCT s) AS studentCount,
       COLLECT(DISTINCT c.title) AS courses
ORDER BY studentCount DESC
LIMIT 1;
